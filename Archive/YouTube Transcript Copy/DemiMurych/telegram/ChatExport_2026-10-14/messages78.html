// Универсальная функция создания строки таблицы
function createTableRow(product, properties) {
    const row = document.createElement('tr');
    row.setAttribute('data-product', JSON.stringify(product));

    const cells = properties.map(prop => `<td>${product[prop] || ''}</td>`).join('');
  // console.log(JSON.parse(row.getAttribute('data-product')));
    row.innerHTML = cells;
    return row;
}


// Добавляем строки в таблицу
products.forEach(product => {
    const row = createTableRow(product, properties);
    tableBody.appendChild(row);
});

Этот код динамически создает строки таблицы HTML, заполняя их данными из массива объектов products. Сначала он получает ссылку на элемент <tbody> таблицы с идентификатором table-body. Затем, используя функцию createTableRow, он создает строки для каждого продукта, заполняя ячейки таблицы значениями, соответствующими свойствам, указанным в массиве properties.

Ключ к коллекциям Map и Set согласно спецификации, формируется согласно абстрактной операции CanonicalizeKeyedCollectionKey. Где и абстрагируется алгоритм формирования ключа. Все ключи для колекций Map Set обязательно проходят через него.

Внутрене представление чисел тут совершенно не причем. Уже хотя бы потому, что оно не имеет однозначного поведения.

В общем случае числа должны вести себя согласно стандарту IEEE754. Ключевое слово должны. Потому что в зависимости от выражения, поведение может радикальным образом изменяться.

Как следствие, мы не можем опираться на стандарт IEEE754 как на тот который справделив везде и всегда в JS. Как следствие невозможносно использовать Number:equals как опорную точку. Как следствие нужно заводить новую абстракцию.

Что было и сделано для коллекций Map и Set. Где добавлено исключительное поведение для -0 +0.

К слову сказать, при StrictEquals это исключение так же добавлено в спецификацию. То есть -0 === +0 // true.

Тем не менее, часть выражений JS напрямую зависят от того что мы имеем +0 или -0.

Например: -0 + 0; // -0
Или Object.is(-0, +0); // false

Так что, было бы конечно отлично, если бы все это зависело исключительно от IEEE754. Но нет.

И это то, почему JS можно называть WEIRD.
