    try {
        const rules = Array.from(sheet.cssRules);
        rules.forEach(rule => {
            if (rule.style && rule.style.backgroundImage) {
                const urlMatch = rule.style.backgroundImage.match(/url$["']?([^"']+)["']?$/);

type StableCallbackRef<T> = React.MutableRefObject<{
  stableCallback: T;
  callback: T;
}>;

// This type is used to constrain argument to a function
// eslint-disable-next-line @typescript-eslint/ban-types
type AnyMethod = Function;

/**
 * same as useCallback, but with zero dependencies, allowing to access
 * the current closure variables without re-creating the function.
 *
 * @param callback A function (callback) that needs to be memoized
 * @returns A function that does the same as callback, but is stable.
 **/
const useStableCallback = <CB extends AnyMethod>(callback: CB): CB => {
  if (typeof callback != 'function') {
    throw new Error('useStableCallback expects a function as an argument');
  }
  const stableCallbackRef: StableCallbackRef<CB> = React.useRef({
    //@ts-expect-error - Using this for noImplicitAny to work
    stableCallback: ((...args) => stableCallbackRef.current.callback(...args)) as CB,
    callback,
  });

  useIsomorphicLayoutEffect(() => {
    stableCallbackRef.current.callback = callback;
  });

  return stableCallbackRef.current.stableCallback;
};

export default useStableCallback;
