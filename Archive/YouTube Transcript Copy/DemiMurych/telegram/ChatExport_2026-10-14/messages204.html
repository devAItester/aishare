📌 Коментарі до реалізації

SameType & Strict Equality
Якщо типи однакові, результат — через IsStrictlyEqual. Це точна семантика === згідно зі спецификацією.

null/undefined
Специфікація визначає, що null == undefined і undefined == null → true.

Типи String ↔ Number та BigInt ↔ String
Проводиться приведення значення String до Number/BigInt із повторним викликом алгоритму.

Boolean
Boolean приводиться до Number перш ніж порівнювати.

Object
Якщо один із аргументів — об’єкт, приводиться до примітивного через ToPrimitive.

BigInt ↔ Number
Специфікація явно обробляє це по-різному — порівнює реальні значення, але повертає false якщо не скінченні (Infinity, NaN).

📌 Обмеження

Реальна спецификаційна функція ToPrimitive має більше нюансів (виклик Symbol.toPrimitive, обробка екзотичних об’єктів і т.д.). У цій реалізації — базова логіка.

Обробка слотів специфічних об’єктів (як [[IsHTMLDDA]]) пропущена — це Normative Optional.

function isLooselyEqual(x, y) {
  // 1. SameType?
  if (typeof x === typeof y) {
    return isStrictlyEqual(x, y);
  }

  // 2–3. null and undefined comparison
  if ((x === null)
     && (y === undefined)) {
    return true;
  }
  if ((x === undefined)
    && (y === null)) {
    return true;
  }

  // 5. Number == String
  if ((typeof x === "number")
    && (typeof y === "string")) {
    return isLooselyEqual(x, Number(y));
  }

  // 6. String == Number
  if ((typeof x === "string")
    && (typeof y === "number")) {
    return isLooselyEqual(Number(x), y);
  }

  // 7. BigInt == String
  if ((typeof x === "bigint")
    && (typeof y === "string")) {
    try {
      const n = BigInt(y);
      return isLooselyEqual(x, n);
    } catch {
      return false;
    }
  }

  // 8. String == BigInt
  if ((typeof x === "string")
    && (typeof y === "bigint")) {
    return isLooselyEqual(y, x);
  }

  // 9. Boolean == anything
  if (typeof x === "boolean") {
    return isLooselyEqual(Number(x), y);
  }

  // 10. anything == Boolean
  if (typeof y === "boolean") {
    return isLooselyEqual(x, Number(y));
  }

  // 11. Primitive == Object
  if (
    (
      [
        "string",
        "number",
        "bigint",
        "symbol"
      ].includes(typeof x)
    )
    && (typeof y === "object")
    && (y !== null)
  ) {
    return isLooselyEqual(x, toPrimitive(y));
  }

  // 12. Object == Primitive
  if ((typeof x === "object")
    && (x !== null)
    && (
      [
        "string",
        "number",
        "bigint",
        "symbol"
      ].includes(typeof y)
    )
  ) {
    return isLooselyEqual(toPrimitive(x), y);
  }

  // 13. BigInt == Number or Number == BigInt
  if (
    ((typeof x === "bigint")
    && (typeof y === "number"))
    ||
    ((typeof x === "number")
    && (typeof y === "bigint"))
  ) {
    // If either is not finite → false
    if (!Number.isFinite(x)
      || !Number.isFinite(y)) {
      return false;
    }
    return Number(x) === Number(y);
  }

  // 14. default to false
  return false;
}

// Strict equality per spec (delegates to ===)
function isStrictlyEqual(x, y) {
  return x === y;
}

// ToPrimitive helper
function toPrimitive(value) {
  if ((value === null)
    || (typeof value !== "object")) {
    return value;
  }
  const prim = value.valueOf();
  if (typeof prim !== "object") {
    return prim;
  }

  return String(value);
}
