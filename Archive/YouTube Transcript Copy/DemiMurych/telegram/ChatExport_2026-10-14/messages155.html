   var listArguments = textFunction.substring(0, startBodyFunction)
   if (listArguments[0]==="(") listArguments = listArguments.substring(1)
   if (listArguments[listArguments.length-1]===")") listArguments = listArguments.substring(0, listArguments.length-1);
   listArguments = listArguments.replaceAll(" ","").split(",");
   textFunction = textFunction.substring(startBodyFunction+2);
   if (listArguments[0]) textFunction = textFunction.replaceAll(listArguments[0], "array[i]");
   if (listArguments[1]) textFunction = textFunction.replaceAll(listArguments[1], "i");
