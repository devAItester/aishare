async function main() {
  const feed = "f_1_0_3_en_1";
  const url = `https://d.flashscorekz.com/x/feed/${feed}`;

  try {
    const response = await axios.get(url, { headers });
    const data = response.data.split("¬");
    const structuredData = {};

    let currentLeague = null;
    let leagueData = {};
    let matchData = {};

    data.forEach((item, index) => {
      const parts = item.split("÷");
      const key = parts[0];
      const value = parts[parts.length - 1];

      // Проверяем, является ли ключ заголовком раздела
      if (key.includes("~")) {
        // Сохраняем данные о предыдущей лиге (если она была)
        if (currentLeague) {
          structuredData[currentLeague] = {
            ...leagueData,
            ...matchData
          };
        }

        currentLeague = value; // Обновляем текущее название лиги
        leagueData = {}; // Обнуляем данные о лиге
        matchData = {}; // Обнуляем данные о матчах
      } else if (FILTERED_KEYS_LEAGUE.includes(key)) {
        leagueData[key] = value; // Заполняем данные о лиге
      } else if (FILTERED_KEYS_MATCH.includes(key)) {
        matchData[key] = value; // Заполняем данные о матче
      }

      // Обрабатываем последний элемент данных
      if (index === data.length - 1 && currentLeague) {
        structuredData[currentLeague] = {
          ...leagueData,
          ...matchData
        };
      }
    });

    const jsonData = JSON.stringify(structuredData, null, 2);
    fs.writeFileSync("scrap.json", jsonData);
    console.log("Данные сохранены в файл scrap.json");
  } catch (error) {
    console.error("Ошибка при получении данных:", error);
  }
}
