const queue ={
    lengths: [-1,0],
    elements:[],
    set(size){
        this.lengths=[-1,0];
        this.elements=[];
        for (var i=0;i<size;i++){
            this.elements.push(0);
        }
    },
    increaseLengthAndCheckOverflow(i){
        if (this.lengths[i]==this.elements.length){
            this.lengths[i]=0;
        } else this.lengths[i]++;
        return !(this.lengths[0]==this.lengths[1]);
    },
    push(value){
        this.elements[this.lengths[1]]=value;
        return this.increaseLengthAndCheckOverflow(1);
    },
    shift(){
        const res=this.increaseLengthAndCheckOverflow(0);
        const value = this.elements[this.lengths[0]];
        return res&&value;
    }
}

Дальше по сути:
В таких языках как JavaScript, вызов функции - это не тоже самое что и вызов функции в языке C или на уровне машинного кода.
Это всего лишь декларация намерений это сделать.

Реализация этих намерений описывается в спецификации. Где, как например в ECMAScript, ничего подобного вызову функции так как это происходит в Си или в типичном x86 процессоре не происходит. (Превед Сеньеру ПЛюс).

В случае спецификации ECMAScript - вызов функции, это свой собственный процесс, где есть шаги похожие на типичный вызов функций, но только похожий. В результате чего, предсказать как это будет - достаточно сложно. Особенно тогда, когда интерпретатор коотрый пользует наш код, пытается это делать эффективно. ( Превед Сеньеру Плюс еще раз)


Дальше вступает в силу особенность реализации кажджого конкретного агента.
Например V8 имеет несколько разных стратегий выполнения функций:
1) от универсальной и характерной для типичного соглашения вызова функций, где стек используется как часть формирования блока вызова функции
2) до ситуаций где стек вообще не имеет никакого отношения к работе, поскольку он там не нужен ( например в случае методов map reduce и так далее, когда они не имееют сайд эффектов)


Идем дальше.
Одна из особенностей языка JavaScript заключается в том, что он позволяет работать в функциональной парадигме. Где вместо цифклов привычно использовать рекурсию.

Однако, при применении традиционной схемы работы с функциями, где каждый вызов ее формирует кадр в стеке для каждого вызова, мы так или иначе упираемся в величину этого стека. (смотрит типичное решение для вычисления чисел фибоначи с рекурсией)

Для этого в спецификации ECMA предусмотрели оптимизацию Tail Call. Которая позволяла бы избегать именно этих проблем ( традиционного формирования фрейма в стеке и call).

В задании есть несколько непонятных для меня моментов. Один из них — обязательное условие использования рекурсии в итераторе. Дело в том, что итератор уже и так содержит встроенные средства для перехода к следующему шагу (вызов метода next). Следовательно, ни рекурсия, ни цикл для этой цели вовсе не нужны. Это обязательное условие имеет не больше смысла, чем задание, требующее написать точно такое же решение с использованием генератора без цикла и рекурсии. Можно конечно же переизобретать свои итераторы/генераторы... но зачем? 🚲

Если проигнорировать обязательное условие, то можно предложить такое решение:
function getFibonacciIterator(n) {
  let prev = 1n, curr = 0n;
  
  return {
    next() {
      const done = --n < 0;
      if (done) return { done };
      const value = curr;
      curr = prev + curr;
      prev = value;
      return { value, done };
    },
    [Symbol.iterator]() {
      return this;
    }
  };
}

const fibonacci = getFibonacciIterator(10000);
const sequence = [...fibonacci];

С генератором аналогичное решение могло бы выглядеть намного лаконичнее:
function* fibonacciGenerator(n) {
  let prev = 1n, curr = 0n;
  
  while (n-- > 0) {
    yield curr;
    [prev, curr] = [curr, prev + curr];
  }
}

const fibonacci = fibonacciGenerator(10000);
const sequence = [...fibonacci];


Второй непонятный момент. Итераторы и генераторы (как удобный способ создания первых) хорошо подходят для задач создания/обхода последовательностей значений. Однако если нам нужно найти только какое-то конкретное значение последовательности, то приспосабливать сюда итераторы может быть не самой лучшей идеей. 🩼

Для решения конкретной задачи лучше не использовать итераторы вовсе, обычной функции будет вполне достаточно.
const fibonacciRecursion = (n) => (
  (function fib(num) {
    if(num === 0)  return { prev: 1n, curr: 0n };
    let { prev, curr } = fib(--num);
    curr = prev + (prev = curr);
    return { prev, curr };
  }(n)).curr
);

const fibonacciLoop = (n) => {
  let prev = 1n, curr = 0n;
  while (n--) {
    curr = prev + (prev = curr);
  }
  return curr;
};


При больших значениях n эту задачу можно ещё эффективнее решить с использованием умножения матриц, но в этот алгоритм пока не углублялся. 🤓

кто-то задавал вопрос почему получается цикличность в прототипах:

function Point(x, y) {
    this.x = x;
    this.y = y;
}
Point.prototype.move = function(x, y) {
    this.x += x;
    this.y += y;
};

Point.prototype.toString = function() {
    return `[${this.x}, ${this.y}]`;
};
const pt = Point.prototype.constructor;
const pt2 = Point.prototype.constructor.prototype.constructor.prototype.constructor;
console.log(pt === pt2); // true
const pt3 = Point.prototype;
const pt4 = Point.prototype.constructor.prototype.constructor.prototype;
console.log(pt3 === pt4); // true
const p1 = new Point(10, 20);

каждый объект имеет свойство constructor (например p1.constructor) - ссылка на функцию-конструктор, которая создала этот объект (тут Point)
функция-конструктор имеет свойство prototype (Point.prototype) - объект, который будет являться прототипоv для создаваемых функцией-конструтором (Point) объектов
Этот объект-прототип (Point.prototype) также имеет своиство constructor (Point.prototype.constructor), которое, опять же, ссылается на функцию-конструтор (Point)
вот поэтому и получается цикличность
все объекты имеют свойство constructor, а все конструкторы (функции) - свойство prototype

Хост-среда управляет макрозадачами и event loop.

1. Завершает стек вызовов.

3. Берет одну макрозадачу.

4. Повторяет.
