import React from 'react';

type Ref<T> = React.MutableRefObject<{
  stableCallback: T;
  callback: T;
}>;

// eslint-disable-next-line @typescript-eslint/no-explicit-any
type Fn = (...args: any[]) => any;

/**
 * same as useCallback, but with zero dependencies, allowing to access
 * the current closure variables without re-creating the function.
 *
 * @param callback A function (callback) that needs to be memoized
 * @returns A function that does essentially the same as callback, but is stable.
 **/
const useStableCallback = <CB extends Fn>(callback: CB): CB => {
  const ref: Ref<CB> = React.useRef({
    stableCallback: ((...args) => ref.current.callback(...args)) as CB,
    callback,
  });

  React.useInsertionEffect(() => {
    ref.current.callback = callback;
  });

  return ref.current.stableCallback;
};

export default useStableCallback;
