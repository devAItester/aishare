Можно сделать класс обёртку над промисом:


class MyPromise {
    _isFulfilled = false;
    _isResolved = false;
    _isRejected = false;

    constructor() {
        this.promise = new Promise((resolve, reject) => {
            this.resolve = value => {
                this._isFulfilled = true;
                this._isResolved = true;
                resolve(value);
            };

            this.reject = error => {
                this._isFulfilled = true;
                this._isRejected = true;
                reject(error);
            };
        });
    }

    get isFulfilled() {
        return this._isFulfilled;
    }

    get isResolved() {
        return this._isResolved;
    }

    get isRejected() {
        return this._isRejected;
    }
}
