var foo = ‘bar’
// 1) пихает в память две записи “стринга с ввелью ‘foo’” и “стринга с велью ‘bar’”

// 1) строки foo и bar уже есть в памяти ( в куче этой вашей ) - v8 не будет память под них вылелять

function whatever(...args){} - вот так можно, в аргументах в описании функции, — оно возьмет всё что пришло в функцию, все аргументы, и сделает из этого массив args, если вызвать whatever(1, 2) - args будет равно [1, 2]

whatever(...arr) - вот так можно, в аргументах при вызове функции, — оно всё что в массиве arr возьмёт и передаст в функцию, как если бы через запятую перечислили - whatever( ...[1, 2] ) — то же самое что whatever(1, 2) или там whatever.apply(this, [1, 2])

{...obj} - вот так можно, в объекте, оно проитерирует все ключи объекта obj и запишет их и их значения в новый объект. При этом для примитивных значений записано в объект ничего не будет ({...false} - валидный синтаксис, как и {...5}, но смысла нет особо. За исключением строк, разумеется, потому что они iterable, хоть и примитивные)

[...arr] - для массивов - тоже валидный синтаксис, оно возьмет элементы массива и создаст из них новый массив

const {...obj2} = obj - вот так можно, при объявлении. Оно создаст из всех ключей и значений obj новый объект obj2

const [...arr2] = arr - при объявлении можно и для массивов (ну, для любых iterable в общем то, если оно не iterable - будет ошибка)
