})();

// Helper function to wait for the file to be downloaded
function waitForDownload(downloadPath) {
    return new Promise((resolve, reject) => {
        fs.watch(downloadPath, (eventType, filename) => {
            if (eventType === 'rename' && filename) {
                resolve(filename);
            }
        });
    });
}
